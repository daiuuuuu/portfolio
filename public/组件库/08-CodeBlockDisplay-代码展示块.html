async def execute_pipeline(img_path: Path):
    # Step 1: Vision Perception
    raw_json = await agent_ocr.recognize(img_path)

    # Step 2: Logical Re-ordering
    reordered_text = await agent_reorder.process(
        input=raw_json,
        context="Yan Fu Logic Translation"
    )

    # Step 3: Semantic Correction
    final_output = await agent_correct.refine(
        text=reordered_text,
        check_vertical_alignment=True
    )

    return final_output
